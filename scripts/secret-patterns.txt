# Secret patterns for scripts/scrub-check.sh (extended regular expressions, one per line).
# Blank lines and lines starting with '#' are ignored. A match fails the check; the
# matched text is never printed, only the file and line.
# Keep patterns specific: a false positive blocks every close.

# AWS access key ids (long-term and temporary)
\b(AKIA|ASIA)[0-9A-Z]{16}\b

# Private key block headers
-----BEGIN ([A-Z0-9]+ )*PRIVATE KEY( BLOCK)?-----

# GitHub tokens (classic and fine-grained)
\bgh[pousr]_[A-Za-z0-9]{36,}\b
\bgithub_pat_[A-Za-z0-9_]{22,}\b

# sk- style API keys (covers sk-ant-... and sk-proj-...)
\bsk-[A-Za-z0-9_-]{20,}

# Slack tokens
\bxox[abprs]-[A-Za-z0-9-]{10,}

# Google API keys
\bAIza[0-9A-Za-z_-]{35}\b
